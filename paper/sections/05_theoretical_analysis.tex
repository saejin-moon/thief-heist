\section{Theoretical Analysis}
\label{sec:theory}

We now establish analytical bounds on the stability of Fisher parameter recombination and the anti-monopoly guarantees of the load-balancing objective.

\paragraph{Fisher Information Geometry and Policy Preservation.}
Let $\mathcal{P} = \{\pi_\theta : \theta \in \Theta\}$ denote the statistical manifold of stochastic policies equipped with the Fisher Information metric tensor $g_{ij}(\theta) = \mathbb{E}_{\pi_\theta} \left[\frac{\partial \log \pi_\theta}{\partial \theta_i} \frac{\partial \log \pi_\theta}{\partial \theta_j}\right]$. Under second-order Taylor expansion around parameter vector $\theta_k$, the Kullback-Leibler divergence between parent policy $\pi_{\theta_k}$ and recombined policy $\pi_\theta$ satisfies:
\begin{equation}
D_{\text{KL}}(\pi_{\theta_k} \parallel \pi_\theta) = \frac{1}{2} (\theta - \theta_k)^\top F_k (\theta - \theta_k) + \mathcal{O}(\|\theta - \theta_k\|^3)
\end{equation}

\begin{prop}[Curvature-Optimal Recombination]
\label{prop:recombination}
Let $\{\theta_k\}_{k=1}^K$ be parameter vectors of parent policies with positive definite Fisher matrices $\{F_k\}_{k=1}^K$ and non-negative mixture weights $w_k$ satisfying $\sum_k w_k = 1$. To second order in $\|\theta - \theta_k\|$, the recombined parameter vector $\theta_{\text{recomb}} = \left(\sum_k w_k F_k\right)^{-1} \left(\sum_k w_k F_k \theta_k\right)$ minimizes the weighted sum of Riemannian distances on the statistical manifold:
\begin{equation}
\theta_{\text{recomb}} = \arg\min_\theta \sum_{k=1}^K w_k D_{\text{KL}}(\pi_{\theta_k} \parallel \pi_\theta) + \mathcal{O}\!\left(\|\theta-\theta_k\|^3\right)
\end{equation}
\end{prop}
\begin{proof}
Differentiating the quadratic approximation $\mathcal{J}(\theta) = \frac{1}{2} \sum_k w_k (\theta - \theta_k)^\top F_k (\theta - \theta_k)$ with respect to $\theta$ yields $\nabla_\theta \mathcal{J}(\theta) = \sum_k w_k F_k (\theta - \theta_k) = 0$. Inverting the positive-definite sum $\sum_k w_k F_k$ yields the closed-form minimizer $\theta_{\text{recomb}}$.
\end{proof}
Proposition~\ref{prop:recombination} guarantees that parameter recombination preserves high-probability action representations from parents in parameter regions where Fisher information is large, preventing catastrophic forgetting of critical sub-skills.

\paragraph{Load-Balancing Anti-Monopoly Bound.}
\begin{prop}[Routing Entropy Lower Bound]
\label{prop:entropy}
\label{prop:entropy}
Let $f_k = \frac{1}{B}\sum_{i=1}^B \mathbb{I}(k_{\text{chosen},i}=k)$ denote the empirical dispatch frequency of expert $k \in \{1,\dots,K\}$ over mini-batch $B$, and $P_k = \frac{1}{B}\sum_{i=1}^B P_{i,k}$ its mean gating probability. Suppose routing is calibrated such that $P_k = f_k$ for all $k$, and the auxiliary load-balancing loss satisfies $\mathcal{L}_{\text{balance}} \le 1 + \delta$ for some $\delta \ge 0$. Then the routing entropy $H(f) = -\sum_k f_k \log f_k$ is strictly bounded away from zero:
\begin{equation}
\label{eq:entropy_bound}
H(f) \ge \log K - \frac{\delta}{2}
\end{equation}
Consequently, as $\delta \to 0$, the routing distribution approaches uniform dispersion $H(f) \to \log K$, precluding single-expert monopoly regimes ($H(f) = 0$).
\end{prop}
\begin{proof}
Under calibration, $\mathcal{L}_{\text{balance}} = K \sum_k f_k^2 \le 1 + \delta$, hence $\sum_k f_k^2 \le \frac{1+\delta}{K}$. The squared Euclidean deviation of the dispatch distribution from the uniform distribution $u_k = 1/K$ is
$\textstyle \sum_k (f_k - 1/K)^2 = \sum_k f_k^2 - \frac{1}{K} \le \frac{\delta}{K}$.
Since the entropy $H$ is $\frac{K}{2}$-strongly concave in the $\ell_2$ norm around $u$---i.e., $H(f) \ge H(u) - \frac{K}{2} \sum_k (f_k - 1/K)^2$---we obtain
$H(f) \ge \log K - \frac{K}{2} \cdot \frac{\delta}{K} = \log K - \frac{\delta}{2}$.
\end{proof}
Proposition~\ref{prop:recombination} and the bound in Eq.~\eqref{eq:entropy_bound} jointly ensure that Fisher recombination preserves parent skills while the balancing objective keeps the routing distribution dispersed: for the configuration used in our experiments ($K \le 8$ observed experts, $\mathcal{L}_{\text{balance}} \le 1 + \delta$ with small $\delta$), entropy remains within $\delta/2$ nats of uniform, so monopoly regimes ($H(f)=0$) require $\delta \ge 2\log K$.
